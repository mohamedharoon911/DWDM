
EXPERIMENT 1 – EQUAL FREQUENCY BINNING AND DATA SMOOTHING

Code:

# Sorted grocery price dataset
data <- c(2,3,3,4,4,4,6,6,6,6,7,7,
          8,9,9,10,10,10,11,11,12,12,12,12,
          13,14,14,15,15,15,16,16,17,18,18,18,
          18,19,20,20,20,21,22,22,23,24,25)

# Divide into four equal-frequency bins
bins <- split(data, ceiling(seq_along(data) / 12))

cat("Equal Frequency Bins:\n")
print(bins)

# Calculate bin means
means <- sapply(bins, mean)
cat("\nBin Means:\n")
print(round(means, 2))

# Smoothing by bin means
smooth_mean <- lapply(bins, function(x) {
  rep(round(mean(x), 2), length(x))
})
cat("\nSmoothing by Bin Means:\n")
print(smooth_mean)

# Smoothing by bin boundaries
smooth_boundary <- lapply(bins, function(x) {
  low <- min(x)
  high <- max(x)
  sapply(x, function(v) {
    if (abs(v - low) <= abs(v - high)) low else high
  })
})
cat("\nSmoothing by Bin Boundaries:\n")
print(smooth_boundary)

# Histogram
hist(data, breaks = 4, col = "skyblue",
     main = "Grocery Price Histogram",
     xlab = "Price")




EXPERIMENT 2 – STATISTICAL ANALYSIS AND BOXPLOT

Code:

# Marks of two sections
A <- c(72,68,81,55,90,74,88,60,77,85)
B <- c(65,70,78,62,69,80,75,73,71,68)

# Section A statistics
cat("SECTION A\n")
cat("Mean:", mean(A), "\n")
cat("Median:", median(A), "\n")
cat("Range:", diff(range(A)), "\n")

# Section B statistics
cat("\nSECTION B\n")
cat("Mean:", mean(B), "\n")
cat("Median:", median(B), "\n")
cat("Range:", diff(range(B)), "\n")

# Combine data and draw boxplot
marks <- data.frame(
  Marks = c(A, B),
  Section = rep(c("A", "B"), each = 10)
)

boxplot(Marks ~ Section, data = marks,
        col = c("skyblue", "orange"),
        main = "Section A and B Comparison",
        ylab = "Marks")
